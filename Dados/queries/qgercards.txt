        LEFT JOIN GERCARDS.CARDS_UNIDADE CU ON C.ID_UNIDADE = CU.ID

SELECT
	CU.ID ,CU.NOME AS NOME_ASSINANTE ,CU.EMAIL AS EMAIL_ASSINANTE,CU.TIPO,CE.NOME_EMPRESA ,CN.NOME_NAVIO,
        C.ID, C.ID_EMPRESA_NAVIO, C.NUMERO_CONTRATO, C.NOME_REQUISITANTE, C.ID_CARDS, C.DESCRICAO, C.DATA_INICIO, C.DATA_TERMINO, C.QUANTIDADE, C.ID_UNIDADE AS UNIDADE,C.UNIDADE_OUTRO,
        C.ENVELOPE_ID, C.DATA_ASSINATURA,C.DATA_SOLICITACAO,CEN.ID_EMPRESA,CEN.ID_NAVIO,TITD.ID_CARDS_TIPO_ID, 
        TITD.ID_CARDS_TIPO_DETALHE_ID, CE.NOME_EMPRESA,CN.NOME_NAVIO, C.DATA_REJEICAO,C.MOTIVO_REJEICAO,U1.NOME AS QUEM_REJEITOU,CU1.SIGLA, CU1.DESCRICAO AS DESCRICAO_UNIDADE  FROM 		
		GERCARDS.CARDS_USUARIO_EMPRESA_NAVIO cuen INNER JOIN GERCARDS.CARDS_USUARIO cu ON CUEN.ID_USUARIO  = CU.ID
		INNER JOIN GERCARDS.CARDS_EMPRESA_NAVIO CEN ON CUEN.ID_EMPRESA_NAVIO = CEN.ID
		INNER JOIN GERCARDS.CARDS c ON C.ID_EMPRESA_NAVIO  = CUEN.ID_EMPRESA_NAVIO
        INNER JOIN GERCARDS.CARDS_EMPRESA CE ON CEN.ID_EMPRESA = CE.ID 
        INNER JOIN GERCARDS.CARDS_NAVIO CN ON CEN.ID_NAVIO = CN.ID 
        INNER JOIN GERCARDS.CARDS_TIPO_ID_TIPO_DETALHE_ID TITD ON C.ID_CARDS_TIPO_ID_DET_ID = TITD.ID
        LEFT JOIN GERCARDS.CARDS_USUARIO U1 ON C.ID_USUARIO_REJEICAO = U1.ID
        LEFT JOIN GERCARDS.CARDS_UNIDADE CU1 ON C.ID_UNIDADE = CU1.ID 
WHERE 
        C.ID = #ID# AND CU.TIPO  IN ('S','A') AND CU.ATIVO = 'S'

[QRY_GET_CARD_POR_USER_ID]
SELECT 
        C.ID, C.ID_EMPRESA_NAVIO, C.NUMERO_CONTRATO, C.NOME_REQUISITANTE, C.ID_CARDS,C.TIPO_OUTRO, C.DESCRICAO, C.DATA_INICIO, C.DATA_TERMINO, TO_CHAR(c.QUANTIDADE, 'FM999999990.000') AS QUANTIDADE, C.ID_UNIDADE AS UNIDADE, C.NOME_ASSINANTE, 
	EMAIL_ASSINANTE, ENVELOPE_ID, DATA_ASSINATURA,DATA_SOLICITACAO,CU.DESCRICAO AS DESCRICAO_UNIDADE,C.DATA_REJEICAO,C.TIPO_REQUISITANTE,
	CE.NOME_EMPRESA,CN.NOME_NAVIO,CU2.NOME AS USUARIO_ARQUIVAMENTO,C.DT_ARQUIVAMENTO  
FROM GERCARDS.CARDS C INNER JOIN GERCARDS.CARDS_EMPRESA_NAVIO EN ON
	C.ID_EMPRESA_NAVIO = EN.ID INNER JOIN GERCARDS.CARDS_EMPRESA CE ON EN.ID_EMPRESA = CE.ID 
	INNER JOIN GERCARDS.CARDS_NAVIO CN ON EN.ID_NAVIO = CN.ID 
	LEFT JOIN GERCARDS.CARDS_UNIDADE CU ON C.ID_UNIDADE = CU.ID 
	LEFT JOIN GERCARDS.CARDS_USUARIO CU2 ON C.ID_USUARIO_ARQUIVAMENTO  = CU2.ID
WHERE 
	EN.ID  IN (
	SELECT ID_EMPRESA_NAVIO FROM GERCARDS.CARDS_USUARIO_EMPRESA_NAVIO WHERE ID_USUARIO =#ID_USUARIO#) AND C.ID=#ID#
